cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(MatcoreArtifactSymbolOwnership LANGUAGES C CXX)
  include(CTest)
  find_package(LLVM 21.1.8 EXACT CONFIG REQUIRED)
  include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/MatcoreDSLCompilerOptions.cmake")
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib/support" support)
endif()
if(NOT COMMAND mdslc_target_match_llvm_rtti)
  function(mdslc_target_match_llvm_rtti target)
    if(NOT LLVM_ENABLE_RTTI)
      target_compile_options(${target} PRIVATE -fno-rtti)
    endif()
  endfunction()
endif()
if(NOT TARGET matcore_artifact_symbol_ownership)
  add_library(matcore_artifact_symbol_ownership STATIC
    ../../lib/codegen/ArtifactSymbolOwnership.cpp)
  target_compile_features(matcore_artifact_symbol_ownership PUBLIC cxx_std_20)
  target_include_directories(matcore_artifact_symbol_ownership PUBLIC ../../lib/codegen)
  target_include_directories(matcore_artifact_symbol_ownership SYSTEM PUBLIC ${LLVM_INCLUDE_DIRS})
  target_link_libraries(matcore_artifact_symbol_ownership PUBLIC LLVMCore
    PRIVATE LLVMObject LLVMTargetParser)
  mdslc_target_enable_warnings(matcore_artifact_symbol_ownership WERROR)
  mdslc_target_match_llvm_rtti(matcore_artifact_symbol_ownership)
  set(artifact_ownership_test_llvm LLVMIRReader LLVMAsmParser)
else()
  # Integrated Clang/MLIR owns one shared LLVM runtime; do not add static LLVM.
  set(artifact_ownership_test_llvm LLVM)
endif()

add_executable(matcore_artifact_symbol_ownership_test artifact_symbol_ownership_test.cpp)
target_link_libraries(matcore_artifact_symbol_ownership_test PRIVATE
  matcore_artifact_symbol_ownership ${artifact_ownership_test_llvm} MatcoreDSL::PlatformSupportV1)
mdslc_target_enable_warnings(matcore_artifact_symbol_ownership_test WERROR)
mdslc_target_match_llvm_rtti(matcore_artifact_symbol_ownership_test)
if(NOT MDSLC_CLANGXX_EXECUTABLE)
  find_program(MDSLC_CLANGXX_EXECUTABLE NAMES clang++-21 REQUIRED)
endif()
add_test(NAME codegen.artifact_symbol_ownership_v1
  COMMAND matcore_artifact_symbol_ownership_test "${MDSLC_CLANGXX_EXECUTABLE}")
set_tests_properties(codegen.artifact_symbol_ownership_v1 PROPERTIES TIMEOUT 180)
