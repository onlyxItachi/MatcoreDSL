cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(MatcoreAuthenticatedHostThunk LANGUAGES C CXX)
  include(CTest)
  find_package(LLVM 21.1.8 EXACT CONFIG REQUIRED)
  include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/MatcoreDSLCompilerOptions.cmake")
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib/support" support)
endif()
if(NOT COMMAND mdslc_target_match_llvm_rtti)
  function(mdslc_target_match_llvm_rtti target)
    if(NOT LLVM_ENABLE_RTTI)
      target_compile_options(${target} PRIVATE -fno-rtti)
    endif()
  endfunction()
endif()

if(NOT TARGET matcore_authenticated_host_thunk)
add_library(matcore_authenticated_host_thunk STATIC
  ../../lib/codegen/AuthenticatedHostThunk.cpp)
target_compile_features(matcore_authenticated_host_thunk PUBLIC cxx_std_20)
target_include_directories(matcore_authenticated_host_thunk PUBLIC ../../lib/codegen)
target_include_directories(matcore_authenticated_host_thunk SYSTEM PUBLIC ${LLVM_INCLUDE_DIRS})
target_link_libraries(matcore_authenticated_host_thunk PUBLIC LLVMCore
  PRIVATE LLVMLinker LLVMBitReader LLVMBitWriter)
mdslc_target_enable_warnings(matcore_authenticated_host_thunk WERROR)
mdslc_target_match_llvm_rtti(matcore_authenticated_host_thunk)
endif()

add_executable(matcore_authenticated_host_thunk_test authenticated_host_thunk_test.cpp)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(thunk_test_llvm LLVMIRReader LLVMAsmParser LLVMTransformUtils)
else()
  # The integrated compiler owns one shared LLVM runtime with Clang. Do not
  # resurrect a second static parser/command-line registry in its test binary.
  set(thunk_test_llvm LLVM)
endif()
target_link_libraries(matcore_authenticated_host_thunk_test PRIVATE
  matcore_authenticated_host_thunk ${thunk_test_llvm} MatcoreDSL::PlatformSupportV1)
mdslc_target_enable_warnings(matcore_authenticated_host_thunk_test WERROR)
mdslc_target_match_llvm_rtti(matcore_authenticated_host_thunk_test)
if("${CMAKE_CXX_FLAGS} ${MDSLC_CLOSED_REGION_CXX_FLAGS}" MATCHES "fsanitize=.*address")
  target_compile_definitions(matcore_authenticated_host_thunk_test PRIVATE MDSLC_ABI_THUNK_SANITIZE=1)
endif()
if(NOT MDSLC_CLANGXX_EXECUTABLE)
  find_program(MDSLC_CLANGXX_EXECUTABLE NAMES clang++-21 REQUIRED)
endif()
add_test(NAME codegen.authenticated_host_thunk_v1
  COMMAND matcore_authenticated_host_thunk_test "${MDSLC_CLANGXX_EXECUTABLE}")
set_tests_properties(codegen.authenticated_host_thunk_v1 PROPERTIES TIMEOUT 180)
